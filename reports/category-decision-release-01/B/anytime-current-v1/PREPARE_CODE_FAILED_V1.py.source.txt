"""One registered current-origin delivery of an already tested binary recipe.

No historical refit/search, acquisition, offers, selector, scheduler or website
publication. Raw CSVs and the training matrix are never exported.
"""
import argparse
import csv
import gzip
import hashlib
import json
import math
import platform
import re
import resource
import sys
import time
from datetime import datetime, timezone
from pathlib import Path

import numpy as np
import sklearn
from threadpoolctl import threadpool_limits

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
from nfl_v2 import empirical_benchmark as empirical
from nfl_v2 import hgb_control as hgb
from nfl_v2.anytime_current_shadow import VERSION, RECIPE, export_model, restore_probability
from tools.capture_qb_forward_timing import digest, read, write_new


def now():
    return datetime.now(timezone.utc).isoformat()


def prepare(args):
    out = args.out.resolve()
    out.mkdir(exist_ok=False, parents=True)
    cutoff = datetime.now(timezone.utc).timestamp()
    games = hgb.schedules(args.source_root / 'games.csv')
    sources, rows, seen = [], [], set()
    need = set(empirical.PHYSICAL_COUNTS) | set(empirical.PHYSICAL_YARDS) | {'rushing_tds', 'receiving_tds'}
    for season in range(2022, 2027):
        path = args.source2026 if season == 2026 else args.source_root / ('stats_player_week_' + str(season) + '.csv')
        sources.append(dict(source_id=path.name, sha256=digest(path), bytes=path.stat().st_size,
            rights='NFLverse ordinary licensed basic-statistical derivative; raw file not republished'))
        with path.open() as f:
            reader = csv.DictReader(f)
            if not need <= set(reader.fieldnames):
                raise ValueError('Exact physical/event source fields missing')
            if {'available', 'outcome_available_at', 'observed_available_at'} & set(reader.fieldnames):
                raise ValueError('Observed/unknown clock route requires separate qualification')
            for raw in reader:
                if raw['season_type'] != 'REG' or int(raw['season']) != season or raw['position'] not in hgb.POSITIONS:
                    continue
                if not re.fullmatch(r'00-\d{7}', raw['player_id']):
                    continue
                game = games.get(raw['game_id'])
                if not game or game['season'] != season or game['week'] != int(raw['week']):
                    raise ValueError('Historical game identity')
                if (raw['team'], raw['opponent_team']) not in ((game['home'], game['away']), (game['away'], game['home'])):
                    raise ValueError('Historical team identity')
                key = raw['game_id'], raw['player_id']
                if key in seen:
                    raise ValueError('Repeated historical identity')
                seen.add(key)
                values = {k: empirical.numeric(raw[k]) for k in need}
                proof = next((dict(field=k, value=values[k], rule='VALID_COUNT_GT0') for k in empirical.PHYSICAL_COUNTS
                    if empirical.valid_count(values[k]) and values[k] > 0), None)
                proof = proof or next((dict(field=k, value=values[k], rule='FINITE_SIGNED_YARDS_NONZERO')
                    for k in empirical.PHYSICAL_YARDS if values[k] is not None and values[k] != 0), None)
                played = proof is not None
                target = empirical.target_value('anytime_td', values) if played else None
                rows.append(dict(season=season, week=int(raw['week']), game_id=raw['game_id'],
                    player_id=raw['player_id'], position=raw['position'], team=raw['team'], opponent=raw['opponent_team'],
                    kickoff=game['kickoff'], available=game['kickoff'] + 7 * 86400,
                    rushing_tds=values['rushing_tds'], receiving_tds=values['receiving_tds'],
                    played=played, physical_proof=proof, values=dict(anytime_td=target)))
    rows.sort(key=lambda r: (r['kickoff'], r['game_id'], r['player_id']))
    roster = read(args.roster)
    basic = []
    ids = set()
    for r in roster:
        if r['team'] not in ('WAS', 'NYG') or r['position'] not in ('RB', 'WR', 'TE'):
            continue
        if r['gsis_id'] in ids or not re.fullmatch(r'00-\d{7}', r['gsis_id']) or r['season'] != 2026:
            raise ValueError('Current canonical roster identity')
        ids.add(r['gsis_id'])
        basic.append(dict(player_id=r['gsis_id'], name=r['full_name'], position=r['position'], team=r['team'],
            opponent='NYG' if r['team'] == 'WAS' else 'WAS', provider_status=r['status'],
            eligible=r['status'] == 'ACT', game_id='2026_10_WAS_NYG', native_event_id='401873052',
            season=2026, week=10, kickoff=1794532500., cutoff=cutoff,
            membership='CURRENT_KNOWN_SOURCE_SNAPSHOT_CONDITIONAL_APPEARANCE',
            individual_transaction_effective=None, expected_starter=None, health=None))
    if not basic or not any(r['eligible'] for r in basic):
        raise ValueError('Empty supported current universe')
    write_new(out / 'EVENT_HISTORY.json.gz', rows)
    write_new(out / 'CURRENT_QUERIES.json', dict(basic=basic, target_outcomes_required=False))
    code = [Path(__file__), ROOT / 'nfl_v2/anytime_current_shadow.py', ROOT / 'nfl_v2/hgb_control.py',
            ROOT / 'nfl_v2/empirical_benchmark.py', ROOT / 'nfl_v2/week8_shadow.py']
    write_new(out / 'FROZEN_PROTOCOL.json', dict(id=VERSION, registered_UTC=now(), fit_cutoff=cutoff,
        recipe=RECIPE, history_regime='H-MULTI', owner='B', source_ids=sources,
        schedule=dict(source_id='games.csv', sha256=digest(args.source_root / 'games.csv')),
        current_roster=dict(repository='NickNicky19/NFL-forecasting-NGS-C',
            commit='d3d1ef4dc60826334234cfa1a1b8e373b2ee024e', path='sources/profile-closure-next-02/roster_current_season.json',
            sha256=digest(args.roster), source_expires_at=1794454469.),
        inputs_sha256=digest(out / 'EVENT_HISTORY.json.gz'), queries_sha256=digest(out / 'CURRENT_QUERIES.json'),
        code_sha256={str(p.relative_to(ROOT)): digest(p) for p in code},
        feature_order=list(hgb.FEATURES), params=hgb.PARAMS,
        training='All eligible QB/RB/WR/TE known binary labels since2022; strict availability<actualcutoff; unknown slots retained in history',
        feature_policy='Exact original mean3/n3/mean8/n8 and4value/age/missing lags; fixed7day reconstruction at all historical/current origins',
        target='At leastone rushingORreceivingTD; explicit component/positivephysicalproof;unknownnotzero',
        control='Unchanged H-MULTI own empirical event frequency, then sameposition mean, then literal0; no fitting/tuning',
        probabilities='Raw event probabilities, NOT calibratedconfidence/EV; no offeredNO/Under invented',
        horizon='REAL_EARLY_CURRENT_SHADOW;originalissue/source/persistenceclocksseparate;notoriginalkickoffhistoricalissuance',
        stop='One current fit once; no history/tuning/search repeats; retain any failed stages',
        budget=dict(max_new_current_fits=1, max_new_historical_fits=0, wall_seconds=300, cpu_seconds=300,
            RSS_bytes=1073741824, threads=2, public_bytes=8388608),
        new_primary_release=False, exact_offer=None, side=None, production_owner='Originalwebsiteownerunchanged'))
    print(json.dumps(dict(state='PREPARED_BEFORE_FIT', history_slots=len(rows), basic=len(basic), eligible=sum(r['eligible'] for r in basic))))


def run(args):
    out = args.out.resolve()
    start, cpu = time.monotonic(), time.process_time()
    p = read(out / 'FROZEN_PROTOCOL.json')
    for path, wanted in p['code_sha256'].items():
        if digest(ROOT / path) != wanted:
            raise ValueError('Code changed')
    if digest(out / 'EVENT_HISTORY.json.gz') != p['inputs_sha256'] or digest(out / 'CURRENT_QUERIES.json') != p['queries_sha256']:
        raise ValueError('Input changed')
    if datetime.now(timezone.utc).timestamp() >= p['current_roster']['source_expires_at']:
        raise ValueError('Current roster source expired')
    rows = read(out / 'EVENT_HISTORY.json.gz')
    cutoff = p['fit_cutoff']
    if any(r['available'] != r['kickoff'] + 7 * 86400 for r in rows):
        raise ValueError('Different history policy')
    history = hgb.FeatureHistory(rows)
    train = [r for r in rows if r['available'] < cutoff and r['values']['anytime_td'] is not None]
    x = np.asarray([history.get(r['player_id'], r['position'], 'anytime_td', r['kickoff'])[0] for r in train])
    y = np.asarray([r['values']['anytime_td'] for r in train])
    if len(y) < 40 or len(np.unique(y)) != 2:
        raise ValueError('Minimum model support not met')
    basic = read(out / 'CURRENT_QUERIES.json')['basic']
    qs = [q for q in basic if q['eligible']]
    qx = np.asarray([history.get(q['player_id'], q['position'], 'anytime_td', cutoff)[0] for q in qs])
    write_new(out / 'FIT_ATTEMPT.json', dict(actual_started_UTC=now(), training_rows=len(y), calls=1))
    with threadpool_limits(limits=2):
        model = hgb.make_estimator('anytime_td', 'log_loss_probability').fit(x, y)
        payload = export_model(model)
        native = model.predict_proba(qx)[:, list(model.classes_).index(1.)]
    write_new(out / 'MODEL.json.gz', dict(model=payload, actual_fit_completed_UTC=now(), training_rows=len(y),
        fit_cutoff=cutoff, training_membership_sha256=hashlib.sha256(hgb.canonical([(r['game_id'], r['player_id']) for r in train])).hexdigest(),
        latest_training_label=max(r['available'] for r in train), protocol_sha256=digest(out / 'FROZEN_PROTOCOL.json')))
    predictions = []
    maximum_error = 0.
    for q, feature, expected in zip(qs, qx, native):
        probability = restore_probability(payload, feature, hgb.FEATURES)
        maximum_error = max(maximum_error, abs(probability - expected))
        own = [r['values']['anytime_td'] for r in train if r['player_id'] == q['player_id'] and r['position'] == q['position']]
        position = [r['values']['anytime_td'] for r in train if r['position'] == q['position']]
        values = own or position
        reference = sum(values) / len(values) if values else 0.
        _, latest = history.get(q['player_id'], q['position'], 'anytime_td', cutoff)
        for method, value in [(RECIPE + ':H-MULTI', probability), ('empirical-player-history-control-v1:H-MULTI', reference)]:
            predictions.append(dict(identity=q, procedure_id=method, target='inherited_rushing_or_receiving_anytime_td',
                product='RAW_EVENT_PROBABILITY', probability=float(value), complement=float(1 - value),
                feature_cutoff=cutoff, latest_feature_available=latest, own_prior_n=len(own),
                fallback='none' if method.startswith('hgb') or own else 'position_empirical' if position else 'constant_zero',
                actual=None, actual_state='FUTURE_NOT_GRADED', offered_sides=None, line=None, selected_side=None,
                confidence_status='UNCALIBRATED_RESEARCH_NOT_PER_PICK_CONFIDENCE_OR_EV'))
    if maximum_error > 1e-12:
        raise ValueError('Trusted JSON restoration differs')
    resources = dict(new_current_fits=1, new_historical_fits=0, wall_seconds=time.monotonic()-start,
        cpu_seconds=time.process_time()-cpu, peak_RSS_bytes=resource.getrusage(resource.RUSAGE_SELF).ru_maxrss*1024)
    if resources['wall_seconds'] > 300 or resources['cpu_seconds'] > 300 or resources['peak_RSS_bytes'] > 1073741824:
        raise ValueError('Registered current-fit budget exceeded')
    write_new(out / 'PREPARED_FORECASTS.json', dict(id=VERSION, prepared_UTC=now(), basic=basic, predictions=predictions,
        resources=resources, model_sha256=digest(out / 'MODEL.json.gz'), parity_max_abs=maximum_error,
        new_offers=0, new_selections=0, new_grades=0, eligible_capture=False,
        reason='Fresh native event admission and real issuance/persistence are separate',
        environment=dict(python=platform.python_version(), numpy=np.__version__, sklearn=sklearn.__version__)))
    print(json.dumps(dict(state='CURRENT_INFERENCE_PREPARED_NOT_ISSUED', forecasts=len(predictions), resources=resources)))


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('action', choices=['prepare', 'run'])
    parser.add_argument('--out', type=Path, required=True)
    parser.add_argument('--source-root', type=Path)
    parser.add_argument('--source2026', type=Path)
    parser.add_argument('--roster', type=Path)
    args = parser.parse_args()
    prepare(args) if args.action == 'prepare' else run(args)
